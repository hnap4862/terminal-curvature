TerminalCurvature.Geometry.matched_tree_every_word (word : List Bool) :
  ∃ c p,
    0 < p ∧
      Family.SignedCounts (Family.coeff c p) word ∧
        Geometry.Forest (Adj (Family.arrows c p)) ∧
          (∀ (u v : Family.Labels c p × Bool), ∃ n ≤ 10, Geometry.Walk (Adj (Family.arrows c p)) u v n) ∧
            (∀ (u : Family.Labels c p × Bool),
                Geometry.mate (Geometry.mate u) = u ∧
                  Geometry.mate u ≠ u ∧ Adj (Family.arrows c p) u (Geometry.mate u)) ∧
              (∀ (S : Finset (Family.Labels c p × Bool)),
                  (Independent (Family.arrows c p) fun x => x ∈ S) → S.card ≤ Fintype.card (Family.Labels c p)) ∧
                ∃ S, (Independent (Family.arrows c p) fun x => x ∈ S) ∧ S.card = Fintype.card (Family.Labels c p)
